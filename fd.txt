public Mono<ActionCollectionDto> getActions(
        String authorization,
        List<String> types
) {
    return Mono.defer(() -> {
        Set<String> allowedTypes = Set.of(
                DOC_TYPE_CASHOUT,
                DOC_TYPE_CASHIN,
                FLAG,
                HINT
        );

        if (types == null || types.isEmpty()) {
            return Mono.error(
                    new IllegalArgumentException("Список типов action не должен быть пустым")
            );
        }

        boolean hasInvalidType = types.stream()
                .anyMatch(type -> type == null || !allowedTypes.contains(type));

        if (hasInvalidType) {
            return Mono.error(
                    new IllegalArgumentException("Передан неизвестный тип action")
            );
        }

        Set<String> requestedTypes = new HashSet<>(types);

        return bankApiBankCashClient.searchActions(authorization)
                .switchIfEmpty(Mono.error(
                        new IllegalStateException("Пустое тело справочника actions")
                ))
                .flatMap(response -> {
                    if (response.getActions() == null) {
                        return Mono.error(
                                new IllegalStateException("Отсутствует actions")
                        );
                    }

                    List<ActionDto> active = response.getActions().stream()
                            .filter(action -> Boolean.TRUE.equals(action.getIsActive()))
                            .toList();

                    if (active.isEmpty()) {
                        return Mono.error(actionTodo(
                                "Пустой активный справочник: ошибка или пустой ответ"
                        ));
                    }

                    if (requestedTypes.contains(DOC_TYPE_CASHOUT)
                            && active.stream().noneMatch(
                                    action -> DOC_TYPE_CASHOUT.equals(action.getType()))) {
                        return Mono.error(actionTodo(
                                "Нет расходных документов: ошибка или только FLAG/HINT"
                        ));
                    }

                    List<ActionDto> actions = active.stream()
                            .filter(action ->
                                    matchesActionType(action.getType(), requestedTypes))
                            .toList();

                    if (actions.isEmpty()) {
                        return Mono.just(
                                ActionCollectionDto.builder()
                                        .actions(List.of())
                                        .build()
                        );
                    }

                    return bankApiBankCashClient.obtainClientSelected(authorization)
                            .switchIfEmpty(Mono.error(
                                    new IllegalStateException(
                                            "Пустой ответ выбранного клиента")
                            ))
                            .flatMap(client ->
                                    evaluateActions(authorization, actions, client));
                });
    });
}

private boolean matchesActionType(
        String actualType,
        Set<String> requestedTypes
) {
    if (requestedTypes.contains(actualType)) {
        return true;
    }

    boolean documentsRequested =
            requestedTypes.contains(DOC_TYPE_CASHOUT)
                    || requestedTypes.contains(DOC_TYPE_CASHIN);

    return documentsRequested
            && (FLAG.equals(actualType) || HINT.equals(actualType));
}
