    public SpmAnalyzeRequest mapApplicationToSpmAnalyzeRequest(Application application) {
        String mdmCode = application.getClientSnapshot().getMdmCode();
        ProductTypeCode productTypeCode = application.getProductTypeCode();

        SpmAnalyzeRequestBody body = SpmAnalyzeRequestBody.builder()
                .channelIndicator(Optional.ofNullable(CHANNEL_INDICATORS.get(application.getChannel()))
                        .orElse(ChannelIndicatorType.OTHER))
                .clientDefinedChannelIndicator(
                        Objects.requireNonNull(CLIENT_DEFINED_CHANNEL_INDICATORS.get(application.getChannel())))
                .identificationData(identificationData(mdmCode, application.getId(), application.getCreatedUserLogin()))
                .autoCreateUserFlag(true)
                .runRiskType(RunRiskType.ALL)
                .deviceManagementRequest(deviceManagementRequest())
                .deviceRequest(applicationDeviceRequest(application))
                .eventData(EventData.builder()
                        .eventType(EventType.CLIENT_DEFINED)
                        .clientDefinedEventType(Objects.requireNonNull(APPLICATION_EVENT_TYPES.get(productTypeCode)))
                        .eventDescription(Objects.requireNonNull(APPLICATION_EVENT_DESCRIPTIONS.get(productTypeCode)))
                        .transactionData(TransactionData.builder()
                                .amount(amountInKopecks(application.getOperationSum()))
                                .myAccountData(accountData(application.getCashoutAccount(),
                                        application.getClientSnapshot().getShortName(),
                                        application.getAccountBranchBik()))
                                .build())
                        .clientDefinedAttributeList(applicationFacts(application).toAttributes())
                        .build())
                .build();

        return SpmAnalyzeRequest.builder()
                // заголовок заявки отличается от РКО/ПКО: xReplyAsyncKafka, без xInitiatorService/xInitiatorHost — как было
                .requestHeader(SpmRequestHeader.builder()
                        .xMdmId(mdmCode)
                        .requestType(REQUEST_TYPE)
                        .xReplyAsyncKafka(true)
                        .build())
                .requestBody(body)
                .build();
    }

    private Facts applicationFacts(Application application) {
        ApplicationFundsRecipient recipient = application.getFundsRecipient();
        DocumentTypeDto documentType = getDocumentType(recipient.getDocumentTypeCode());
        BankSalePointDto salePoint = getSalePoint(application.getSalePoint().getSalePointCode());
        String recipientFullName = fullName(recipient.getLastName(), recipient.getFirstName(), recipient.getMiddleName());

        Facts facts = new Facts()
                .add("ID сессии", application.getCreatedUserSessionId())
                .add("ИНН плательщика", application.getClientSnapshot().getInn())
                .add("БИК плательщика", application.getAccountBranchBik())
                .add("Код подразделения", application.getBankSalePointBisCode())
                .add("Планируемая дата операции", DATE_TO_SPM_STRING_FORMATTER.format(application.getOperationDate()))
                .add("SystemId", TBCC_INTS)
                .add("Номер документа", application.getNumber())
                // УДАЛЕНО: закомментированный факт "Тип подписания"
                .add("Система источник", SUBO_SOURCE_SYSTEM)
                .add("Бал.счет плательщика", balanceAccount(application.getCashoutAccount()))
                .add("Бал.счет плательщика 1-го порядка", firstOrderBalanceAccount(application.getCashoutAccount()))

                .add("БИК клиента", application.getAccountBranchBik())
                .add("Идентификатор заявки", application.getId().toString())
                .add("Название ТП", salePoint.getTpShortName())
                .add("Наименование банка отправителя", application.getAccountBranchName())
                .add("Номер счета плательщика", application.getCashoutAccount())
                .add("Сумма операции в рублях", rubles(application.getOperationSum()))
                .add("Тип ДУЛ доверенного лица", documentType.getDescription())
                .add("Наименование плательщика", application.getClientSnapshot().getFullName())
                .add("Назначение платежа", APPLICATION_PAYMENT_PURPOSE)

    /** Нет даты — warn и пустое значение, неверный формат — исключение (как было). */
    private static String applicationRecipientBirthDate(String birthDate) {
        try {
            Optional<String> spmDate = toSpmDate(birthDate);
            if (spmDate.isEmpty()) {
                log.warn("Отсутствует дата рождения плательщика");
            }
            return spmDate.orElse(null);
        } catch (DateTimeParseException e) {
            throw new BankCashBusinessException("Дата рождения плательщика в неверном формате " + birthDate);
        }
    }

    // ИЗМЕНЕНО: было mapCashoutToSpmAnalyzeInternally
    private SpmAnalyzeRequest mapCashoutToSpmAnalyzeRequest(Cashout cashout, String employeeLogin) {
        String mdmCode = cashout.getCashoutClientSnapshot().getMdmCode();
        ProductTypeCode eventProductType = cashoutEventProductType(cashout.getProductTypeCode());
        String recipientFullName = cashoutRecipientFullName(cashout);
        String branchBik = Optional.ofNullable(cashout.getAccountBranchInfo()).map(AccountBranchInfo::getBik).orElse(null);

        SpmAnalyzeRequestBody body = SpmAnalyzeRequestBody.builder()
                .channelIndicator(ChannelIndicatorType.BRANCH)
                .clientDefinedChannelIndicator(SUBO)
                .identificationData(identificationData(mdmCode, cashout.getId(), employeeLogin))
                .autoCreateUserFlag(true)
                .runRiskType(RunRiskType.ALL)
                .deviceManagementRequest(deviceManagementRequest())
                .eventData(EventData.builder()
                        .eventType(EventType.WITHDRAW)
                        .clientDefinedEventType(CASHOUT_EVENT_TYPES.get(eventProductType))
                        .eventDescription(CASHOUT_EVENT_DESCRIPTIONS.get(eventProductType))
                        .transactionData(branchTransaction(cashout.getOperationSum())
                                .otherAccountData(accountData(null, recipientFullName, branchBik))
                                .myAccountData(accountData(cashout.getCashoutAccount(),
                                        sanitize(cashout.getCashoutClientSnapshot().getShortName(), NAME_MAX_LENGTH),
                                        branchBik))
                                .build())
                        .clientDefinedAttributeList(cashoutFacts(cashout).toAttributes())
                        .build())
                .build();

        return SpmAnalyzeRequest.builder()
                .requestHeader(documentHeader(mdmCode))
                .requestBody(body)
                .build();
    }

    private Facts cashoutFacts(Cashout cashout) {
        ProductTypeCode productTypeCode = cashout.getProductTypeCode();
        var recipient = cashout.getCashoutFundsRecipient();
        BankBranchesDto accountBankBranch = refTablesClient.getBankBranchByCftCode(cashout.getCashoutAccountAbsBranchCode());
        BankSalePointDto salePoint = getSalePoint(cashout.getBankSalePointBisCode());
        DocumentTypeDto documentType = getDocumentType(recipient.getDocumentTypeCode());
        String recipientFullName = cashoutRecipientFullName(cashout);

        Facts facts = new Facts()
                .add("ИНН плательщика", cashout.getCashoutClientSnapshot().getInn())
                .add("Код подразделения", salePoint.getBranchCode())
                .add("Название ТП", sanitize(cashout.getCashoutSalePoint().getTpShortName(), TEXT_MAX_LENGTH))
                .add("Наименование получателя", recipientFullName)
                .add("Сведения о ДУЛ доверенного лица", dulInfo(recipient.getDocSeries(), recipient.getDocNumber()))
                .add("ФИО доверенного лица", recipientFullName)
                .add("ФИО сотрудника", sanitize(fullName(cashout.getSentToControlLastName(),
                        cashout.getSentToControlFirstName(), cashout.getSentToControlMiddleName()), NAME_MAX_LENGTH))
                .add("SystemId", TBCC_INTS)
                .add("Номер документа", productTypeCode == ProductTypeCode.CASHWITHOUTCHECK
                        ? String.valueOf(cashout.getRkoNumber())
                        : String.join(" ", cashout.getCheckSeries(), cashout.getCheckNumber()))
                .add("Наименование банка отправителя", sanitize(accountBankBranch.getName(), TEXT_MAX_LENGTH))
                .add("Назначение платежа", sanitize(cashout.getCashoutPurpose(), TEXT_MAX_LENGTH))
                .add("Наименование плательщика", sanitize(cashout.getCashoutClientSnapshot().getShortName(), NAME_MAX_LENGTH))
                .add("Система источник", SUBO_SOURCE_SYSTEM)
                .add("Бал.счет плательщика", balanceAccount(cashout.getCashoutAccount()))
                .add("Бал.счет плательщика 1-го порядка", firstOrderBalanceAccount(cashout.getCashoutAccount()))
                .add("БИК плательщика", accountBankBranch.getBik())
                .add("БИК (SWIFT) плательщика", accountBankBranch.getBik())
                .add("БИК клиента", accountBankBranch.getBik())
                .add("Номер счета плательщика", cashout.getCashoutAccount())
                .add("Сумма операции в рублях", rubles(cashout.getOperationSum()))
                .add("Тип ДУЛ доверенного лица", documentType.getDescription())
                .add("Логин сотрудника, создавшего операцию", cashout.getSentToControlEmployeeLogin())
                .add("Страна ДУЛ доверенного лица", cashoutRecipientIssueCountry(cashout))
                .add("Планируемая дата операции", cashout.getOperationDate().format(DATE_TO_SPM_STRING_FORMATTER));

        if (productTypeCode == ProductTypeCode.CASHWITHOUTCHECK) {
            // TODO уточнить: для РКО по заявке в "Идентификатор заявки" уходит id РКО, а не id заявки. Оставлено как было.
            facts.add("Идентификатор заявки", String.valueOf(cashout.getId()));
        }
        return addCashSymbols(facts, cashout.getSymbols(), s -> s.getCashSymbolCode(), s -> s.getSymbolSum());
    }

    /** Реестровая выдача без чека описывается в СПМ как CASHWITHOUTCHECK. */
    private static ProductTypeCode cashoutEventProductType(ProductTypeCode productTypeCode) {
        return productTypeCode == ProductTypeCode.REGISTRYAPPCASHWITHOUTCHECK
                ? ProductTypeCode.CASHWITHOUTCHECK
                : productTypeCode;
    }

    private static String cashoutRecipientFullName(Cashout cashout) {
        var recipient = cashout.getCashoutFundsRecipient();
        return sanitize(fullName(recipient.getLastName(), recipient.getFirstName(), recipient.getMiddleName()),
                NAME_MAX_LENGTH);
    }

    /**
     * ИСПРАВЛЕНО (замечание ПСИ, CASH_WITH_REQUEST BRANCH DENY AML: "Страна ДУЛ доверенного лица — отсутствует value").
     * У РКО по заявке cashout_funds_recipient.issue_country_code не заполнен — берём страну ДУЛ
     * из получателя связанной заявки (то же поле, что уходит в СПМ по самой заявке).
     * TODO по-хорошему копировать страну при создании РКО из заявки (cashoutMapper.toCashoutWithApplication),
     *  тогда этот fallback не нужен. Связь application должна быть загружена (@EntityGraph), иначе LazyInitializationException.
     */
    private static String cashoutRecipientIssueCountry(Cashout cashout) {
        String issueCountry = cashout.getCashoutFundsRecipient().getIssueCountryCode();
        if (StringUtils.isBlank(issueCountry) && cashout.getApplication() != null) {
            return cashout.getApplication().getFundsRecipient().getIssueCode();
        }
        return issueCountry;
    }

    public SpmAnalyzeRequest mapCashinOvnPkoNotaryDepToSpmAnalyzeRequest(Cashin cashin) {
        var recipient = cashin.getCashinFundsRecipient();

        // проверки до формирования запроса — в том же порядке, что были
        DocumentTypeDto documentType = getDocumentType(recipient.getDocumentTypeCode());
        // ИЗМЕНЕНО: результат getCountry не использовался (CountryDto countryDto) — оставлена только проверка,
        // что страна есть в справочнике; в факт по-прежнему уходит код страны
        requireCountryExists(recipient.getIssueCountryCode());
        String recipientBirthDate = cashinRecipientBirthDate(recipient.getBirthDate());
        if (cashin.getAccountBranchInfo() == null || StringUtils.isEmpty(cashin.getAccountBranchInfo().getBik())) {
            throw new BankCashBusinessException("Отсутствуют данные по филиалу получателя");
        }
        if (cashin.getSalePointBranchInfo() == null || StringUtils.isEmpty(cashin.getSalePointBranchInfo().getBik())) {
            throw new BankCashBusinessException("Отсутствуют данные по филиалу банка отправителя");
        }

        boolean isOvn = ProductTypeCode.OVN.equals(cashin.getProductTypeCode());
        String mdmCode = cashin.getCashinClientSnapshot().getMdmCode();

        SpmAnalyzeRequestBody body = SpmAnalyzeRequestBody.builder()
                .channelIndicator(ChannelIndicatorType.BRANCH)
                .clientDefinedChannelIndicator(SUBO)
                .identificationData(identificationData(mdmCode, cashin.getId(), frkkSessionSecurityService.getEmployeeLogin()))
                .autoCreateUserFlag(false) // TODO у заявки и РКО true — уточнить, намеренно ли
                .runRiskType(RunRiskType.ALL)
                .deviceManagementRequest(deviceManagementRequest())
                .eventData(EventData.builder()
                        .eventType(EventType.DEPOSIT)
                        .clientDefinedEventType(isOvn ? CASH_OVN_DEP : CASH_PKO_NOTARY_DEP)
                        .eventDescription(isOvn ? CASH_OVN_DESCRIPTION : CASH_PCO_NOTARY_DESCRIPTION)
                        .transactionData(branchTransaction(cashin.getOperationSum())
                                .otherAccountData(accountData(cashin.getCashinAccount(),
                                        sanitize(cashin.getCashinClientSnapshot().getFullName(), NAME_MAX_LENGTH),
                                        cashin.getAccountBranchInfo().getBik()))
                                .myAccountData(accountData(null,
                                        cashinRecipientFullName(cashin),
                                        cashin.getSalePointBranchInfo().getBik()))
                                .build())
                        .clientDefinedAttributeList(cashinFacts(cashin, documentType, recipientBirthDate).toAttributes())
                        .build())
                .build();

        return SpmAnalyzeRequest.builder()
                .requestHeader(documentHeader(mdmCode))
                .requestBody(body)
                .build();
    }

    private static Facts cashinFacts(Cashin cashin, DocumentTypeDto documentType, String recipientBirthDate) {
        var recipient = cashin.getCashinFundsRecipient();
        String recipientFullName = cashinRecipientFullName(cashin);
        String receiverBik = cashin.getAccountBranchInfo().getBik();

        Facts facts = new Facts()
                .add("SystemId", TBCC_INTS)
                .add("Номер документа", String.valueOf(cashin.getDocNumber()))
                .add("Назначение платежа", sanitize(cashin.getCashinPurpose(), TEXT_MAX_LENGTH))
                .add("Наименование плательщика", recipientFullName)
                .add("ИНН плательщика", cashin.getCashinClientSnapshot().getInn())
                .add("Сведения о ДУЛ доверенного лица", dulInfo(recipient.getDocSeries(), recipient.getDocNumber()))
                .add("ФИО доверенного лица", recipientFullName)
                .add("Дата рождения доверенного лица", recipientBirthDate)
                .add("Тип ДУЛ доверенного лица", documentType.getDescription())
                .add("Страна ДУЛ доверенного лица", recipient.getIssueCountryCode())
                .add("БИК получателя", receiverBik)
                .add("Наименование банка получателя", sanitize(cashin.getAccountBranchInfo().getName(), TEXT_MAX_LENGTH))
                .add("Наименование получателя", sanitize(cashin.getCashinClientSnapshot().getFullName(), NAME_MAX_LENGTH))
                .add("ИНН получателя", cashin.getCashinClientSnapshot().getInn())
                .add("Номер счета получателя", cashin.getCashinAccount())
                .add("Бал.счет получателя", balanceAccount(cashin.getCashinAccount()))
                .add("Бал.счет получателя 1-го порядка", firstOrderBalanceAccount(cashin.getCashinAccount()))
                .add("Код подразделения", cashin.getBankSalePointBisCode())
                .add("Название ТП", sanitize(cashin.getBankSalePointName(), TEXT_MAX_LENGTH))
                .add("Сумма операции в рублях", rubles(cashin.getOperationSum()))
                .add("Система источник", SUBO_SOURCE_SYSTEM)
                .add("БИК клиента", receiverBik)
                .add("БИК (SWIFT) получателя", receiverBik)
                .add("Логин сотрудника, создавшего операцию", cashin.getSentToControlEmployeeLogin());

        return addCashSymbols(facts, cashin.getSymbols(), s -> s.getCashSymbolCode(), s -> s.getSum());
    }

    private static String cashinRecipientFullName(Cashin cashin) {
        var recipient = cashin.getCashinFundsRecipient();
        return sanitize(fullName(recipient.getLastName(), recipient.getFirstName(), recipient.getMiddleName()),
                NAME_MAX_LENGTH);
    }

    /** Дата обязательна: нет даты или неверный формат — исключение (как было). */
    private static String cashinRecipientBirthDate(String birthDate) {
        try {
            return toSpmDate(birthDate)
                    .orElseThrow(() -> new BankCashDepositorException("Отсутствует дата рождения плательщика"));
        } catch (DateTimeParseException e) {
            throw new BankCashDepositorException("Дата рождения плательщика в неверном формате " + birthDate);
        }
    }

    /** Операция в отделении (РКО, ПКО): сумма + немедленное внутреннее исполнение. Счета добавляет вызывающий. */

                .build();
    }

    // =============================== значения фактов ===============================

    /**
     * Список фактов clientDefinedAttributeList в порядке добавления.
     * ИЗМЕНЕНО: вместо List.of(...) + new ArrayList<>(...) + ручного stream в ClientDefinedAttribute в каждом методе.
     */
    private static final class Facts {

        private final List<Fact> facts = new ArrayList<>();

        Facts add(String name, String value) {
            facts.add(Fact.builder()
                    .name(name)
                    .value(value)
                    .dataType(DataType.STRING)
                    .build());
            return this;
        }

        List<ClientDefinedAttribute> toAttributes() {
            return facts.stream()
                    .map(fact -> ClientDefinedAttribute.builder().fact(fact).build())
                    .collect(Collectors.toList());
        }
    }

    /**
     * Один символ — "Кассовый символ" = код; несколько — "Кассовый символ и сумма" = "код_сумма;код_сумма".
     * ИЗМЕНЕНО: было три реализации (у заявки инлайн, у РКО и ПКО — toCashSymbolsString(Cashout/Cashin)).
     * ИЗМЕНЕНО: у заявки для одного символа без кода подставлялось "CASH_SYMBOL" — заглушка убрана,
     *  теперь как у РКО и ПКО (код символа как есть).
     */
    private static <T> Facts addCashSymbols(Facts facts, Collection<T> symbols,
                                            Function<T, String> code, Function<T, BigDecimal> sum) {
        if (symbols == null || symbols.isEmpty()) {
            return facts;
        }
        if (symbols.size() == 1) {
            return facts.add("Кассовый символ", code.apply(symbols.iterator().next()));
        }
        String codesWithSums = symbols.stream()
                .map(symbol -> String.join("_", code.apply(symbol), sum.apply(symbol).toPlainString()))
                .collect(Collectors.joining(";"));
        return facts.add("Кассовый символ и сумма", codesWithSums);
    }

    /**
     * "Фамилия Имя Отчество" без пустых частей.
     * ИСПРАВЛЕНО: у РКО и ПКО был String.join(" ", фамилия, имя, отчество) — при отсутствии отчества
     *  в СПМ уходило "Иванов Иван null". У заявки при пустом имени получался двойной пробел.
     */
    private static String fullName(String lastName, String firstName, String middleName) {
        return Stream.of(lastName, firstName, middleName)
                .filter(StringUtils::isNotBlank)
                .collect(Collectors.joining(" "));
    }

    /**
     * Серия (без пробелов) + номер ДУЛ.
     * ИСПРАВЛЕНО: при отсутствии номера уходило "...null", теперь пустая строка.
     */
    private static String dulInfo(String docSeries, Object docNumber) {
        return StringUtils.defaultString(docSeries).replaceAll("\\s", "") + Objects.toString(docNumber, "");
    }

    private static String rubles(BigDecimal operationSum) {
        return operationSum.setScale(2, RoundingMode.HALF_UP).toPlainString();
    }

    /** Балансовый счёт второго порядка — первые 5 цифр счёта. */
    private static String balanceAccount(String account) {
        return account.substring(0, 5);
    }

    /** Балансовый счёт первого порядка — первые 3 цифры счёта. */
    private static String firstOrderBalanceAccount(String account) {
        return account.substring(0, 3);
    }

    /** yyyy-MM-dd[Thh:mm[:ss]] -> dd.MM.yyyy; пусто, если даты нет. */
    private static Optional<String> toSpmDate(String birthDate) {
        if (StringUtils.isEmpty(birthDate)) {
            return Optional.empty();
        }
        return Optional.of(LocalDate.parse(birthDate, BIRTH_DATE_TO_DATE_FORMATTER).format(DATE_TO_SPM_STRING_FORMATTER));
    }

    /**
     * Убирает кавычки и обрезает до maxLength символов (по code point'ам).
     * ИЗМЕНЕНО: было formatStringForSAVOBSK — переименовано, логика та же.
     */
    private static String sanitize(String value, int maxLength) {
        return Optional.ofNullable(value).orElse("").replaceAll("[`'\"]", "")
                .codePoints()
                .limit(maxLength)
                .collect(() -> new StringBuilder(maxLength), StringBuilder::appendCodePoint, StringBuilder::append)
                .toString();
    }

    }

    // ИЗМЕНЕНО: было getCountry(...) с неиспользуемым результатом
    private void requireCountryExists(String issueCountryCode) {
        boolean exists = refTablesClient.getCountry().getCountries().stream()
                .map(CountryDto::getCode)
                .anyMatch(code -> code.equals(issueCountryCode));
        if (!exists) {
            throw new BankCashDepositorException(String.format(COUNTRY_NOT_FOUND, issueCountryCode));
        }
    }
